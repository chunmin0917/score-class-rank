function isInspectableUrl(url) {
  if (!url) return false;
  const restrictedSchemes = [
    'chrome://',
    'chrome-extension://',
    'edge://',
    'edge-extension://',
    'about:',
    'file://',
    'data:',
    'javascript:',
    'blob:'
  ];
  return !restrictedSchemes.some((scheme) => url.startsWith(scheme));
}
